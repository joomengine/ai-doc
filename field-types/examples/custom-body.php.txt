	/**
	 * Render Joomla's text input followed by an escaped value preview.
	 *
	 * @return  string
	 */
	protected function getInput()
	{
		$preview = htmlspecialchars((string) $this->value, ENT_QUOTES, 'UTF-8');

		return parent::getInput()
			. '<p class="form-text"><code>' . $preview . '</code></p>';
	}
